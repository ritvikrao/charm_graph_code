# and on top of the source hold, so the two contributions can be read apart.

# before step 7.3 made adaptive bucketing the default.

fold --combine off --batch-fold on --bucket-policy fixed
hold --combine hold --batch-fold off --bucket-policy fixed
hold+fold --combine hold --batch-fold on --bucket-policy fixed
